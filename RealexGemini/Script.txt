saveCur = function()
    if not Safe.writefile or not Cur.id then return end
    local title = Cur.title
    if title == "Новый чат" then
        for _, m in ipairs(Cur.msgs) do
            if m.role == "user" and m.parts and m.parts[1] and m.parts[1].text then
                title = (m.parts[1].text:gsub("%s+", " ")):sub(1, 40)
                if title ~= "" then break end
            end
        end
    end
    Cur.title = title
    pcall(function() Safe.writefile(histPath(Cur.id), HttpService:JSONEncode({ id = Cur.id, title = title, updated = os.time(), messages = Cur.msgs })) end)
    for _, e in ipairs(HistIndex) do
        if e.id == Cur.id then e.title = title; e.updated = os.time(); e.count = #Cur.msgs end
    end
    saveHistIndex()
end

local B64 = "ABCDEFGHIJKLMNOPQRSTUVWXYZabcdefghijklmnopqrstuvwxyz0123456789+/"
local function base64(data)
    local out = {}
    local i = 1
    while i <= #data do
        local b1, b2, b3 = data:byte(i, i + 2)
        local w = (b1 or 0) * 65536 + (b2 or 0) * 256 + (b3 or 0)
        local c1 = math.floor(w / 262144) % 64
        local c2 = math.floor(w / 4096) % 64
        local c3 = math.floor(w / 64) % 64
        local c4 = w % 64
        out[#out + 1] = B64:sub(c1 + 1, c1 + 1) .. B64:sub(c2 + 1, c2 + 1) .. B64:sub(c3 + 1, c3 + 1) .. B64:sub(c4 + 1, c4 + 1)
        i = i + 3
    end
    local pd = (3 - (#data % 3)) % 3
    local s = table.concat(out)
    if pd > 0 then s = s:sub(1, #s - pd) .. string.rep("=", pd) end
    return s
end

local Scan = { remotes = {}, workspaceTop = {}, scripts = {}, tags = {}, gui = {}, player = {}, digest = "", stamp = "" }
local function fullPath(inst)
    local parts = {}
    local cur = inst
    while cur and cur ~= game do parts[#parts + 1] = cur.Name; cur = cur.Parent end
    return table.concat(parts, ".")
end
local function resolvePath(path)
    local segs = {}
    for s in tostring(path):gmatch("[^%.]+") do segs[#segs + 1] = s end
    if segs[1] == "game" then table.remove(segs, 1) end
    local cur = game
    for i, s in ipairs(segs) do
        if i == 1 then cur = game:FindService(s) or game:FindFirstChild(s) else cur = cur and cur:FindFirstChild(s) end
        if not cur then return nil end
    end
    return cur
end

    for _, root in ipairs({ ReplicatedFirst, ReplicatedStorage, Workspace }) do
        for _, d in ipairs(root:GetDescendants()) do

                Scan.remotes[#Scan.remotes + 1] = { path = fullPath(d), class = d.ClassName, name = d.Name }

        end
    end
    for _, d in ipairs(Workspace:GetChildren()) do
        Scan.workspaceTop[#Scan.workspaceTop + 1] = { path = "Workspace." .. d.Name, class = d.ClassName, kids = #d:GetChildren(), attrs = d:GetAttributes() }

    local scriptSource = Safe.getscripts and Safe.getscripts() or nil
    if scriptSource then
        for _, s in ipairs(scriptSource) do
            if s:IsA("LocalScript") or s:IsA("ModuleScript") then Scan.scripts[#Scan.scripts + 1] = { path = fullPath(s), class = s.ClassName, inst = s } end
        end
    else
        for _, root in ipairs({ ReplicatedFirst, ReplicatedStorage, Workspace, Players }) do

local TOOLS = {}
local TOOL_DECLS = {}
local function isEmpty(t) return type(t) ~= "table" or next(t) == nil end
local function defTool(name, description, parameters, fn)
    local decl = { name = name, description = description }
    if parameters then
        local p = { type = parameters.type or "OBJECT" }
        if not isEmpty(parameters.properties) then p.properties = parameters.properties end
        if not isEmpty(parameters.required) then p.required = parameters.required end
        decl.parameters = p
    end
    TOOLS[name] = { fn = fn }
    TOOL_DECLS[#TOOL_DECLS + 1] = decl
end
defTool("list_children", "List direct children of any instance in the live game DataModel.", { type = "OBJECT", properties = { path = { type = "STRING", description = "Dot path" }, depth = { type = "INTEGER", description = "1 or 2" } }, required = { "path" } }, function(args)
    local inst = resolvePath(args.path)
    if not inst then return { error = "path not found: " .. tostring(args.path) } end
    local depth = math.clamp(tonumber(args.depth) or 1, 1, 2)
    local out = {}
    for i, c in ipairs(inst:GetChildren()) do
        if i > 80 then out[#out + 1] = { name = "...", class = "truncated" } break end
        local entry = { name = c.Name, class = c.ClassName, children = #c:GetChildren() }
        if depth == 2 then
            local grand = {}
            for j, g in ipairs(c:GetChildren()) do if j > 15 then grand[#grand + 1] = "..." break end grand[#grand + 1] = g.Name .. " (" .. g.ClassName .. ")" end
            entry.sub = grand
        end
        out[#out + 1] = entry
    end
    return { path = args.path, class = inst.ClassName, count = #inst:GetChildren(), children = out }
end)
defTool("inspect_instance", "Get class, attributes and key properties of one instance.", { type = "OBJECT", properties = { path = { type = "STRING", description = "Dot path" } }, required = { "path" } }, function(args)
    local inst = resolvePath(args.path)
    if not inst then return { error = "path not found: " .. tostring(args.path) } end
    local props = {}
    for _, pn in ipairs({ "Position", "Size", "Health", "MaxHealth", "Text", "ActionText", "ObjectText", "Value", "Anchored", "CanCollide" }) do
        local ok, v = pcall(function() return inst[pn] end)
        if ok and v ~= nil then props[pn] = (typeof(v) == "Vector3" and ("(" .. math.floor(v.X) .. ", " .. math.floor(v.Y) .. ", " .. math.floor(v.Z) .. ")")) or tostring(v) end
    end
    local attrs = {}
    for k, v in pairs(inst:GetAttributes()) do attrs[k] = tostring(v) end
    return { path = args.path, class = inst.ClassName, name = inst.Name, children = #inst:GetChildren(), properties = props, attributes = attrs }
end)

geminiRaw = function(contents, callback, useTools)
    if not Safe.request then callback(nil, "Нет request() — executor не поддерживает HTTP") return end
    if Settings.apiKey == "" then callback(nil, "Не задан API-ключ (Settings, поле внизу)") return end
    local url = "https://generativelanguage.googleapis.com/v1beta/models/" .. Settings.model .. ":generateContent?key=" .. Settings.apiKey
    local body = { contents = contents, systemInstruction = { parts = { { text = Settings.systemPrompt } } }, generationConfig = { temperature = Settings.temperature } }
    if useTools ~= false then body.tools = { { functionDeclarations = TOOL_DECLS } } end
    task.spawn(function()
        local ok, resp = pcall(function() return Safe.request({ Url = url, Method = "POST", Headers = { ["Content-Type"] = "application/json" }, Body = HttpService:JSONEncode(body) }) end)
        if not ok then callback(nil, "HTTP error: " .. tostring(resp)) return end
        if resp.StatusCode == 429 then callback(nil, "Rate limit (429). Смени модель.") return end
        if resp.StatusCode ~= 200 then
            local ok2, data = pcall(function() return HttpService:JSONDecode(resp.Body) end)
            callback(nil, "API " .. resp.StatusCode .. ": " .. (ok2 and data.error and data.error.message or resp.Body:sub(1, 200)))
            return
        end
        local ok3, data = pcall(function() return HttpService:JSONDecode(resp.Body) end)
        if not ok3 then callback(nil, "Битый JSON ответа") return end
        if not data.candidates or not data.candidates[1] then callback(nil, "Пустой ответ: " .. (data.promptFeedback and data.promptFeedback.blockReason or "unknown")) return end
        callback(data.candidates[1].content, nil)
    end)
end

    local o = Instance.new(class)
    for k, v in pairs(props or {}) do

local function padding(parent, l, t, r, b) return new("UIPadding", { PaddingLeft = UDim.new(0, l), PaddingTop = UDim.new(0, t), PaddingRight = UDim.new(0, r), PaddingBottom = UDim.new(0, b) }, parent) end
local function accentGradient(parent, rotation) return new("UIGradient", { Color = ColorSequence.new(C.Accent1, C.Accent2), Rotation = rotation or 45 }, parent) end
local function tween(obj, time, props, style) pcall(function() TweenService:Create(obj, TweenInfo.new(time, style or Enum.EasingStyle.Quint, Enum.EasingDirection.Out), props):Play() end) end
local function interactive(btn, normalT, hoverT)
    local s = new("UIScale", { Scale = 1 }, btn)
    btn.MouseEnter:Connect(function() tween(btn, 0.2, { BackgroundTransparency = hoverT }); tween(s, 0.2, { Scale = 1.05 }) end)
    btn.MouseLeave:Connect(function() tween(btn, 0.2, { BackgroundTransparency = normalT }); tween(s, 0.2, { Scale = 1 }) end)
    btn.MouseButton1Down:Connect(function() tween(s, 0.1, { Scale = 0.94 }) end)
    btn.MouseButton1Up:Connect(function() tween(s, 0.15, { Scale = 1.05 }) end)
end

local function splitCodeBlocks(text)
    local blocks, rest, pos = {}, {}, 1
    while true do
        local startP, endP, lang, code = text:find("```(%w*)%s*(.-)```", pos, false)
        if not startP then break end
        if startP > pos then rest[#rest + 1] = text:sub(pos, startP - 1) end
        blocks[#blocks + 1] = { lang = (lang == "" and "code" or lang), code = code }
        pos = endP + 1
    end
    if pos <= #text then rest[#rest + 1] = text:sub(pos) end
    return blocks, table.concat(rest, ""):gsub("^%s+", ""):gsub("%s+$", "")
end

addMessage = function(text, isUser, layoutOrder, file)
    local _, bubble = makeRow(isUser, layoutOrder)
    new("UIListLayout", { Padding = UDim.new(0, 8), SortOrder = Enum.SortOrder.LayoutOrder }, bubble)
    local lo = 1
    if file then fileCard(bubble, file, isUser and "light" or nil, nil, lo); lo = lo + 1 end
    local blocks, plain = splitCodeBlocks(text or "")
    if plain ~= "" then
        local label = new("TextLabel", { LayoutOrder = lo, AutomaticSize = Enum.AutomaticSize.XY, Size = UDim2.fromOffset(0, 0), BackgroundTransparency = 1, Text = plain, TextWrapped = true, TextColor3 = isUser and C.Dark or C.Text, TextSize = 14, Font = FONT, TextXAlignment = Enum.TextXAlignment.Left, ZIndex = 4 }, bubble)
        new("UISizeConstraint", { MaxSize = Vector2.new(math.huge, math.huge) }, label)
        lo = lo + 1
    end
    if #blocks > 0 and not isUser then
        for _, blk in ipairs(blocks) do
            makeCodeCard(bubble, blk, lo)
            lo = lo + 1
        end
    elseif #blocks > 0 and isUser then
        for _, blk in ipairs(blocks) do
            local lbl = new("TextLabel", { LayoutOrder = lo, AutomaticSize = Enum.AutomaticSize.XY, Size = UDim2.fromOffset(0, 0), BackgroundTransparency = 1, Text = "```" .. blk.lang .. "\n" .. blk.code .. "\n```", TextWrapped = true, TextColor3 = C.Dark, TextSize = 14, Font = FONT_CODE, TextXAlignment = Enum.TextXAlignment.Left, ZIndex = 4 }, bubble)
            new("UISizeConstraint", { MaxSize = Vector2.new(math.huge, math.huge) }, lbl)
            lo = lo + 1
        end
    end
    toBottom()
    return bubble
end

        corner(d, 4)

        while alive do
            for _, d in ipairs(dots) do
                if not alive then break end

agentRun = function(contents, onLog, finalCb)
    local rounds = 0
    local hadCalls = false
    local function finish(texts)
        local ans = table.concat(texts, "\n")
        if (ans:gsub("%s", "") == "") and hadCalls then
            local extra = {}
            for _, c in ipairs(contents) do extra[#extra + 1] = c end
            extra[#extra + 1] = { role = "user", parts = { { text = "Дай финальный ответ текстом на русском, кратко резюмируя найденное. НЕ вызывай функции." } } }
            geminiRaw(extra, function(content2, err2)
                if err2 then finalCb(ans, nil); return end
                local p2 = content2 and content2.parts or {}
                local t2 = {}
                for _, pp in ipairs(p2) do if pp.text then t2[#t2 + 1] = pp.text end end
                finalCb(table.concat(t2, "\n"), nil)
            end, false)
        else
            finalCb(ans, nil)
        end
    end
    local function step(cur)
        geminiRaw(cur, function(content, err)
            if err then finalCb(nil, err) return end
            local parts = content and content.parts or {}
            local texts, calls = {}, {}
            for _, p in ipairs(parts) do
                if p.text then texts[#texts + 1] = p.text end
                local fc = p.functionCall or p.function_call
                if fc then calls[#calls + 1] = fc end
            end
            if #calls == 0 or rounds >= Settings.maxRounds then finish(texts); return end
            hadCalls = true
            rounds = rounds + 1
            cur[#cur + 1] = { role = "model", parts = parts }
            local respParts = {}
            for _, call in ipairs(calls) do
                local argsStr = pcall(function() return HttpService:JSONEncode(call.args or {}) end) and HttpService:JSONEncode(call.args or {}) or ""
                onLog("[tool] " .. tostring(call.name) .. "(" .. argsStr:sub(1, 80) .. ")")
                local tool = TOOLS[call.name]
                local payload
                if tool then local ok, res = pcall(tool.fn, call.args or {}); payload = ok and res or { error = tostring(res) } else payload = { error = "unknown tool: " .. tostring(call.name) } end
                if type(payload) ~= "table" then payload = { result = tostring(payload) } end
                onLog((payload.error and "   [warn] " .. tostring(payload.error):sub(1, 60) or "   [ok]"))
                respParts[#respParts + 1] = { functionResponse = { name = call.name, response = payload } }
            end
            cur[#cur + 1] = { role = "user", parts = respParts }
            step(cur)
        end, true)
    end
    step(contents)
end

local function renderReaderList(filter)
    for _, c in ipairs(rList:GetChildren()) do if c:IsA("TextButton") then c:Destroy() end end
    local data = Scan[currentCat] or {}
    filter = (filter or ""):lower()
    for _, item in ipairs(data) do
        if filter == "" or item.path:lower():find(filter, 1, true) then
            local b = new("TextButton", { Size = UDim2.new(1, -4, 0, 22), BackgroundColor3 = C.White, BackgroundTransparency = 0.92, Text = "  " .. item.path, TextColor3 = C.Sub, Font = FONT, TextSize = 11, TextXAlignment = Enum.TextXAlignment.Left, BorderSizePixel = 0, TextTruncate = Enum.TextTruncate.AtEnd, AutoButtonColor = false, ZIndex = 4 }, rList)
            corner(b, 6)
            b.MouseButton1Click:Connect(function()
                local out = { "PATH:  " .. item.path, "CLASS: " .. item.class }
                if item.kids then out[#out + 1] = "CHILDREN: " .. item.kids end
                if item.value ~= nil then out[#out + 1] = "VALUE: " .. tostring(item.value) end
                if item.attrs and next(item.attrs) then out[#out + 1] = "ATTRIBUTES:"; for k, v in pairs(item.attrs) do out[#out + 1] = "  " .. k .. " = " .. tostring(v) end end
                if item.inst then
                    out[#out + 1] = ""; out[#out + 1] = "== SOURCE =="
                    if Safe.decompile then local ok, src = pcall(function() return Safe.decompile(item.inst) end); out[#out + 1] = ok and tostring(src):sub(1, 12000) or "(decompile failed)" else out[#out + 1] = "(нет decompile)" end
                end
                if currentCat == "remotes" then
                    out[#out + 1] = ""; out[#out + 1] = "== FIRE TEMPLATE =="
                    local segs = {}
                    for s in item.path:gmatch("[^%.]+") do segs[#segs + 1] = s end
                    local codeLine = "local remote = game:GetService(\"" .. segs[1] .. "\")"
                    for i = 2, #segs do codeLine = codeLine .. ":WaitForChild(\"" .. segs[i] .. "\")" end
                    out[#out + 1] = codeLine
                    out[#out + 1] = item.class == "RemoteEvent" and "remote:FireServer(...)" or "remote:InvokeServer(...)"
                end
                rDetail.Text = table.concat(out, "\n")
            end)
        end
    end
end

            window.Position = UDim2.new(startPos.X.Scale, startPos.X.Offset + d.X, startPos.Y.Scale, startPos.Y.Offset + d.Y)
        end
    end)
end
